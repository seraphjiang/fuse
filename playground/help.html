OpenSearch Examples

-- Error search
SELECT service, status, message
FROM cluster_a.application_logs
WHERE status >= 500 LIMIT 20
-- Error count by service
SELECT service, count(*) as errors
FROM cluster_a.application_logs
WHERE status >= 500
GROUP BY service ORDER BY errors DESC
-- PPL: Aggregate by status
source = cluster_a.application_logs
| stats count() by status
| sort - count()

S3 Examples

-- Browse S3 logs
SELECT * FROM s3_o11y.fuse_logs LIMIT 20
-- Join OpenSearch + S3
SELECT l.trace_id, l.service, s.message
FROM cluster_a.application_logs l
JOIN s3_o11y.fuse_logs s ON l.trace_id = s.trace_id
WHERE l.status >= 500

DynamoDB Examples

-- Scan table
SELECT * FROM dynamodb.users LIMIT 20
-- Filter by partition key
SELECT * FROM dynamodb.sessions
WHERE user_id = 'u-12345'

PostgreSQL Examples

-- Query with aggregation
SELECT department, count(*) as cnt
FROM postgres.employees
GROUP BY department ORDER BY cnt DESC
-- Join PostgreSQL + OpenSearch
SELECT p.name, l.status, l.message
FROM postgres.services p
JOIN cluster_a.application_logs l ON p.service_id = l.service

Cross-Datasource Examples

-- Unified log view (UNION ALL)
SELECT _datasource, service, count(*) as errors
FROM cluster_a.application_logs
WHERE status >= 500
GROUP BY _datasource, service
UNION ALL
SELECT _datasource, service, count(*) as errors
FROM cluster_b.application_logs
WHERE status >= 500
GROUP BY _datasource, service
ORDER BY errors DESC
-- Cross-cluster trace correlation (JOIN)
SELECT a.trace_id, a.service, b.service as b_service
FROM cluster_a.application_logs a
JOIN cluster_b.application_logs b ON a.trace_id = b.trace_id
WHERE a.status >= 500 LIMIT 20
-- CTE with cross-source
WITH errors AS (
  SELECT service, count(*) as cnt
  FROM cluster_a.application_logs
  WHERE status >= 500
  GROUP BY service
)
SELECT * FROM errors WHERE cnt > 5
